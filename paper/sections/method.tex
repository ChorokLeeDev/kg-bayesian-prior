\subsection{The Limitation of GP Variance}

\begin{proposition}[Relation-Agnostic]
\label{prop:relation_agnostic}
GP-KGE uncertainty (Eq.~\ref{eq:gp_uncertainty}) is relation-agnostic: for any entity $e$ and relations $r_1, r_2$,
\begin{equation}
    U_{\text{GP}}(e, r_1, \cdot) = U_{\text{GP}}(e, r_2, \cdot)
\end{equation}
\end{proposition}

\begin{proof}
Immediate from Eq.~\ref{eq:gp_uncertainty}: the relation $r$ does not appear in the uncertainty computation.
\end{proof}

This limitation is fundamental, not an implementation detail (see Figure~\ref{fig:gp_limitation}).
Making GP variance relation-aware would require learning $|\mathcal{E}| \times |\mathcal{R}| \times d$ parameters---infeasible for typical KG sizes.

\subsection{Structural Uncertainty via Coverage}

I introduce \emph{coverage} as a relation-specific uncertainty signal.

\begin{definition}[Coverage]
For entity $e$ and relation $r$:
\begin{equation}
    c(e, r) = \mathbb{1}\left[\exists (h', r, t') \in \mathcal{T} : h' = e \lor t' = e\right]
\end{equation}
\end{definition}

Coverage equals 1 if entity $e$ appears with relation $r$ in training, else 0.

\begin{definition}[Structural Uncertainty]
\begin{equation}
    U_{\text{cov}}(h, r, t) = 2 - c(h, r) - c(t, r)
\end{equation}
\end{definition}

This ranges from 0 (both entities seen with $r$) to 2 (neither seen).

\begin{theorem}[Coverage AUROC]
\label{thm:coverage_auroc}
Under random tail corruption with relation sparsity $s_r = 1 - |\{e : c(e,r)=1\}|/|\mathcal{E}|$, let $a = P(\text{both covered} | \text{ID})$ and $b = P(\text{exactly one covered} | \text{ID})$. Then:
\begin{equation}
    \text{AUROC}_{\text{cov}} = \frac{1}{2}\left(a(1+s_r) + s_r \cdot b\right)
\end{equation}
\end{theorem}

\begin{proof}[Proof sketch]
ID uncertainty distribution: $P(U=0)=a$, $P(U=1)=b$, $P(U=2)=1-a-b$.
OOD has head from ID, random tail: $P(U=0)=1-s_r$, $P(U=1)=s_r$.
AUROC follows from $P(U_{\text{ID}} < U_{\text{OOD}}) + \frac{1}{2}P(U_{\text{ID}} = U_{\text{OOD}})$.
Full derivation in Appendix~\ref{app:proof}.
\end{proof}

\subsection{Complementarity of Signals}

\begin{theorem}[Complementarity]
\label{thm:complementarity}
Neither coverage nor GP variance is sufficient alone. Specifically:
\begin{enumerate}
    \item $\exists$ cases where coverage is correct and GP is wrong
    \item $\exists$ cases where GP is correct and coverage is wrong
\end{enumerate}
\end{theorem}

\begin{proof}
\textbf{Case 1 (Coverage correct, GP wrong):} Entity $e$ appears frequently overall (low $\sigma^2_e$) but never with relation $r$.
For triple $(?, r, e)$: Coverage correctly assigns high uncertainty; GP incorrectly assigns low uncertainty.

\textbf{Case 2 (GP correct, Coverage wrong):} An OOD triple $(h, r, t')$ where the randomly sampled tail $t'$ is a rare entity (high $\sigma^2_{t'}$) that happens to have appeared once with relation $r$.
Coverage incorrectly assigns low uncertainty (suggests ID); GP correctly flags the rare entity.
\end{proof}

\subsection{CAGP: Combining Both Signals}

The complementarity theorem motivates combining both signals:

\begin{definition}[CAGP Uncertainty]
\begin{equation}
    U_{\text{CAGP}}(h, r, t) = \alpha \cdot \tilde{U}_{\text{GP}}(h, r, t) + (1-\alpha) \cdot U_{\text{cov}}(h, r, t)
\end{equation}
where $\tilde{U}_{\text{GP}}$ is GP uncertainty normalized to match coverage scale, and $\alpha \in [0,1]$ is a learnable mixing coefficient.
\end{definition}

Implementation details:
\begin{itemize}
    \item Coverage matrix is precomputed from training data (no additional learning)
    \item $\alpha$ is parameterized as $\sigma(\alpha_{\text{logit}})$ to ensure $\alpha \in (0,1)$
    \item Normalization: $\tilde{U}_{\text{GP}} = U_{\text{GP}} \cdot \frac{\mathbb{E}[U_{\text{cov}}]}{\mathbb{E}[U_{\text{GP}}]}$
\end{itemize}

The algorithm adds minimal overhead: one $|\mathcal{E}| \times |\mathcal{R}|$ binary matrix (sparse in practice) and one scalar parameter.
