\subsection{Preliminaries}

\textbf{Frequency}: $\freq(e)$ counts triples containing entity $e$.
\textbf{Coverage}: $c(e,r) \in \{0,1\}$ indicates whether $e$ appears with $r$ in training.

\begin{definition}[OOD Partition]
\label{def:ood}
Fix threshold $\tau > 0$. OOD queries partition into:
\begin{itemize}
\item \textbf{Emerging entities} $\mathcal{D}_{\text{emerge}}$: $\min(\freq(h), \freq(t)) \leq \tau$.
\item \textbf{Novel contexts} $\mathcal{D}_{\text{novel}}$: $\min(\freq(h), \freq(t)) > \tau$ and ($c(h,r)=0$ or $c(t,r)=0$).
\end{itemize}
We set $\tau$ to the 25th percentile of entity frequencies; results vary $<$0.02 AUROC across $\tau \in [5\text{th}, 30\text{th}]$ percentile (Appendix~\ref{app:implementation}).
\end{definition}

Novel-context detection by $U_{\text{str}}$ is perfect by construction since the same $c(e,r)$ defines both the OOD label and the detector. The contribution is the \emph{impossibility} of entity-level detection (Theorem~\ref{thm:impossibility}) and the \emph{prevalence} of novel contexts (11--25\% of test triples); see Appendix~\ref{rem:circularity}.

\subsection{Uncertainty Signals}

\textbf{Semantic uncertainty} (entity embedding variance): $U_{\text{sem}}(h,r,t) = \frac{1}{2}(\bar{\sigma}^2_h + \bar{\sigma}^2_t)$ where $\bar{\sigma}^2_e = \frac{1}{d}\sum_{j=1}^{d} \exp(\ell_{e,j})$ is the mean diagonal variance of the entity embedding posterior. This is a standard relation-agnostic variational embedding.

\textbf{Structural uncertainty} (coverage): $U_{\text{str}}(h,r,t) = 2 - c(h,r) - c(t,r) \in \{0, 1, 2\}$, a deterministic rule with no learned parameters. Crucially, $U_{\text{sem}}$ is \emph{relation-agnostic}: it depends only on entity variances, not the query relation.

\subsection{Theoretical Results}

\begin{theorem}[Impossibility of Relation-Agnostic Detection]
\label{thm:impossibility}
Let $U(h,r,t) = f(\sigma^2_h, \sigma^2_t)$ where $\sigma^2_e$ depends only on entity $e$, and $f$ is any combining function. Under assumptions A1--A3 (Appendix~\ref{app:proof}\footnote{A1 validated: Spearman $\rho(\text{freq}, \sigma^2) \leq {-}0.68$ on all datasets (Table~\ref{tab:assumptions}). A3 validated: $\geq$98\% of novel-context triples have frequency-matched ID counterparts (Appendix~\ref{app:assumption_verification}).}):
\begin{equation}
    \text{AUROC}(U, \mathcal{D}_{\text{novel}}) \leq \frac{1}{2} + O(\epsilon)
\end{equation}
where $\epsilon$ measures frequency-matching tightness (A3).
\end{theorem}

\textbf{Proof sketch.} By A1, $\sigma^2_e$ is determined by $\text{freq}(e)$. By A3, novel-context triples have frequency-matched ID counterparts, so $U(h,r,t) \approx U(h',r',t')$, giving AUROC $\approx 1/2$. Full proof in Appendix~\ref{app:proof}. $\square$

\emph{Scope and limitations.} The bound formally covers any $f(\sigma^2_h, \sigma^2_t)$, including all variational embedding methods that produce entity-level variances. Observed AUROC $< 1/2$ (e.g., 0.40 in Table~\ref{tab:complementarity}) reflects anti-prediction: high-frequency novel-context entities have \emph{lower} variance than the ID average. Ensemble methods (MC Dropout, Deep Ensembles) and energy scoring do \emph{not} formally satisfy A1---their uncertainty depends on the specific triple, not just entity statistics---so the theorem does not logically apply to them. However, their novel-context AUROC is empirically weak on all benchmarks (Table~\ref{tab:complementarity}), suggesting their uncertainties are entity-dominated in practice. We explicitly label this an \emph{impossibility result for the variance-based class} and an \emph{empirical observation for the broader class}; we do not claim formal impossibility beyond the theorem's stated scope. See Appendix~\ref{rem:theorem_structure}.

\begin{proposition}[Complementarity of Uncertainty Signals]
\label{prop:complementarity}
Under assumptions A1--A6 (Appendix~\ref{app:proof}):
\textbf{(i)} Semantic uncertainty achieves AUROC $\approx \frac{1}{2} + O(\epsilon)$ on novel contexts.
\textbf{(ii)} Structural uncertainty achieves AUROC $< 1$ on emerging entities.
\textbf{(iii)} The gain of $U_{\text{comb}} = \alpha U_{\text{sem}} + (1{-}\alpha) U_{\text{str}}$ over $U_{\text{str}}$ is:
\begin{equation}
\text{AUROC}(U_{\text{comb}}) - \text{AUROC}(U_{\text{str}}) \approx \pi_e \cdot \left(A_{\text{emerge}}^{\text{comb}}-A_{\text{emerge}}^{\text{str}}\right)
\label{eq:synergy}
\end{equation}
where $\pi_e$ is the emerging-entity fraction. This is positive whenever $U_{\text{sem}}$ separates emerging entities better than $U_{\text{str}}$ (A6). \emph{Caveat:} A4 ($\Delta < 1$) is a sufficient condition for the novel-context cancellation in Part~(iii). It is violated on all benchmarks ($\Delta \geq 1.0$; Table~\ref{tab:assumptions}). The proposition therefore serves as \emph{directional theory}: the qualitative predictions (semantic helps on emerging, structural helps on novel, combination outperforms either) hold on all datasets and all seeds without exception. Eq.~\ref{eq:synergy} matches observation within 0.002 (Appendix~\ref{app:mixture}). We present the proposition to explain \emph{why} the combination works; it does not provide worst-case guarantees.
\end{proposition}

\textbf{Proof sketch.} (i) follows from Theorem~\ref{thm:impossibility}. (ii) Emerging entities may have coverage for the query relation, making some indistinguishable from ID via coverage alone. (iii) Since structural AUROC on novel contexts $= 1$ by construction, the novel-context terms cancel and the gain is dominated by emerging entities. Full proof in Appendix~\ref{app:proof}.

\subsection{CAGP}
\label{sec:cagp}

We propose CAGP, a minimal realization of the decomposition framework:
\begin{equation}
    U_{\text{CAGP}}(h,r,t) = \alpha \cdot \tilde{U}_{\text{sem}}(h,r,t) + (1-\alpha) \cdot U_{\text{str}}(h,r,t)
    \label{eq:cagp}
\end{equation}
where $\tilde{U}_{\text{sem}}$ is normalized semantic uncertainty and $\alpha{=}0.5$ is a default mixing weight. We use 0.5 because it makes no prior assumption about signal importance; Appendix~\ref{app:alpha_sensitivity} shows AUROC varies $<$0.02 across $\alpha \in [0.05, 0.95]$, making the choice non-critical. Learning $\alpha$ from the link prediction objective is not possible (coverage is discrete and receives no gradient from $\mathcal{L}_{\text{BCE}}$; see Appendix~\ref{app:alpha_sensitivity}). The normalization $\tilde{U}_{\text{sem}} = U_{\text{sem}} \cdot (\bar{U}_{\text{str}} + \epsilon) / (\bar{U}_{\text{sem}} + \epsilon)$ makes semantic uncertainty a rank-preserving tiebreaker \emph{within} coverage strata, yielding +8--12pp emerging-entity gains over $U_{\text{str}}$ alone.

\paragraph{Training.}
Entity embeddings use \emph{reparameterization sampling}~\citep{kingma2014autoencoding}: $\mathbf{e} = \boldsymbol{\mu}_e + \boldsymbol{\sigma}_e \odot \boldsymbol{\epsilon}$, $\boldsymbol{\epsilon} \sim \mathcal{N}(0, I)$, ensuring BCE loss provides variance gradients and preventing KL-driven collapse. The loss combines three terms:
\begin{equation}
\mathcal{L} = \mathcal{L}_{\text{BCE}} + \beta \cdot \mathcal{L}_{\text{KL}} + w_{\text{unc}} \cdot \mathcal{L}_{\text{margin}}
\label{eq:loss}
\end{equation}
with $\beta{=}0.001$, $w_{\text{unc}}{=}0.1$, margin $m{=}0.3$. Ablation confirms $\mathcal{L}_{\text{margin}}$ contributes $<$0.4pp (\S\ref{sec:experiments}).

\paragraph{Coverage matrix.}
We precompute $\mathbf{C} \in \{0,1\}^{|\mathcal{E}| \times |\mathcal{R}|}$ from training data:
$c(e,r) = 1$ iff $\exists (h,r',t) \in \mathcal{T}_{\text{train}}: (e{=}h \lor e{=}t) \land r'{=}r$.
Precomputation is a single pass over training triples ($O(|\mathcal{T}|)$); inference requires two hash lookups per triple ($O(1)$, $<$2\% overhead; Appendix~\ref{app:implementation}).

\paragraph{Design rationale.}
CAGP is intentionally simple---a fixed scalar combination. The contribution is identifying the \emph{necessity} of structural uncertainty (Theorem~\ref{thm:impossibility}) and its \emph{complementarity} with semantic uncertainty (Proposition~\ref{prop:complementarity}), not architectural novelty.

\paragraph{Alternative realization: RelCondVar.}
\label{sec:relcondvar}
The decomposition framework admits richer realizations. We also evaluate \emph{relation-conditioned variance} (RelCondVar): we train per-relation variance parameters $\sigma^2(e, r)$ alongside entity embeddings, where $\sigma^2(e,r)$ is optimized only on training triples containing entity $e$ with relation $r$. When entity $e$ has never appeared with $r$ (zero coverage), $\sigma^2(e,r)$ remains at its initialization (high uncertainty), naturally recovering the coverage signal in a continuous form. On dense-relation KGs (FB15k-237, 237 relations), RelCondVar outperforms CAGP (+1pp overall, +4pp emerging); on sparse-relation KGs (WN18RR with 11 relations, YAGO with 37), insufficient per-relation training data prevents learning meaningful conditioned variances, and RelCondVar reduces to $U_{\text{str}}$. This pattern validates the decomposition framework while revealing a scalability axis: more relations $\Rightarrow$ more signal for learned structural uncertainty. \emph{When to use which:} RelCondVar is preferable on dense-relation KGs ($\geq$100 relations) where per-relation variance can be learned reliably; CAGP is preferable on sparse-relation KGs and as a universal default requiring zero hyperparameter tuning. The shared contribution is the decomposition framework itself---both realizations confirm that structural uncertainty is necessary.
