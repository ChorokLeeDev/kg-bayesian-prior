\subsection{Preliminaries}

\textbf{Frequency}: $\freq(e)$ counts triples containing entity $e$.
\textbf{Coverage}: $c(e,r) \in \{0,1\}$ indicates whether $e$ appears with $r$ in training.

\begin{definition}[OOD Partition]
\label{def:ood}
Fix threshold $\tau > 0$. OOD queries partition into:
\begin{itemize}
\item \textbf{Emerging entities} $\mathcal{D}_{\text{emerge}}$: $\min(\freq(h), \freq(t)) \leq \tau$.
\item \textbf{Novel contexts} $\mathcal{D}_{\text{novel}}$: $\min(\freq(h), \freq(t)) > \tau$ and ($c(h,r)=0$ or $c(t,r)=0$).
\end{itemize}
We set $\tau$ to the 25th percentile of entity frequencies; results vary $<$0.02 AUROC across $\tau \in [5\text{th}, 30\text{th}]$ percentile (Appendix~\ref{app:implementation}).
\end{definition}

Novel-context detection by $U_{\text{str}}$ is perfect by construction since the same $c(e,r)$ defines both the OOD label and the detector. The contribution is the \emph{impossibility} of entity-level detection (Theorem~\ref{thm:impossibility}) and the \emph{prevalence} of novel contexts (11--25\% of test triples); see Remark~\ref{rem:circularity}.

\subsection{Uncertainty Signals}

\textbf{Semantic uncertainty} (entity embedding variance): $U_{\text{sem}}(h,r,t) = \frac{1}{2}(\bar{\sigma}^2_h + \bar{\sigma}^2_t)$ where $\bar{\sigma}^2_e = \frac{1}{d}\sum_{j=1}^{d} \exp(\ell_{e,j})$ is the mean diagonal variance of the entity embedding posterior. This is a standard relation-agnostic variational embedding.

\textbf{Structural uncertainty} (coverage): $U_{\text{str}}(h,r,t) = 2 - c(h,r) - c(t,r) \in \{0, 1, 2\}$, a deterministic rule with no learned parameters. Crucially, $U_{\text{sem}}$ is \emph{relation-agnostic}: it depends only on entity variances, not the query relation.

\subsection{Theoretical Results}

\begin{theorem}[Impossibility of Relation-Agnostic Detection]
\label{thm:impossibility}
Let $U(h,r,t) = f(\sigma^2_h, \sigma^2_t)$ where $\sigma^2_e$ depends only on entity $e$, and $f$ is any combining function. Under assumptions A1--A3 (Appendix~\ref{app:proof}\footnote{A1 validated: Spearman $\rho(\text{freq}, \sigma^2) \leq {-}0.68$ on all datasets (Table~\ref{tab:assumptions}). A3 validated: $\geq$98\% of novel-context triples have frequency-matched ID counterparts (Appendix~\ref{app:assumption_verification}).}):
\begin{equation}
    \text{AUROC}(U, \mathcal{D}_{\text{novel}}) \leq \frac{1}{2} + O(\epsilon)
\end{equation}
where $\epsilon$ measures frequency-matching tightness (A3).
\end{theorem}

\textbf{Proof sketch.} By A1, $\sigma^2_e$ is determined by $\text{freq}(e)$. By A3, novel-context triples have frequency-matched ID counterparts, so $U(h,r,t) \approx U(h',r',t')$, giving AUROC $\approx 1/2$. Full proof in Appendix~\ref{app:proof}. $\square$

\begin{definition}[Relation-Agnostic Embedding]
\label{def:relation_agnostic}
An embedding function $\phi: \mathcal{E} \to \mathbb{R}^d$ is \emph{relation-agnostic} if the learned representation $\phi(e)$ satisfies:
\begin{equation}
I(\phi(e); c(e,r) \mid \freq(e)) = 0 \quad \forall r \in \mathcal{R}
\label{eq:ra_condition}
\end{equation}
where $I(\cdot;\cdot\mid\cdot)$ denotes conditional mutual information. Equivalently: knowing $\phi(e)$ provides no information about whether $e$ has been observed with relation $r$, beyond what entity frequency already reveals.
\end{definition}

\begin{theorem}[Embedding-Based Impossibility]
\label{thm:embedding_impossibility}
Let $\phi: \mathcal{E} \to \mathbb{R}^d$ be a relation-agnostic embedding (Definition~\ref{def:relation_agnostic}), and let $\psi: \mathcal{R} \to \mathbb{R}^k$ be any relation embedding. For any uncertainty estimator $U(h,r,t) = F(\phi(h), \psi(r), \phi(t))$ where $F$ is measurable:
\begin{equation}
\text{AUROC}(U, \mathcal{D}_{\text{novel}}) \leq \frac{1}{2} + O(\epsilon)
\end{equation}
under assumptions A1 and A3 (Appendix~\ref{app:proof}), where $\epsilon$ measures frequency-matching tightness.
\end{theorem}

\textbf{Proof sketch.}
By Definition~\ref{def:relation_agnostic}, $\phi(e)$ is conditionally independent of $c(e,r)$ given $\freq(e)$. Thus any function $F(\phi(h), \psi(r), \phi(t))$ can distinguish novel contexts from ID only via frequency differences. By A3, novel-context triples have frequency-matched ID counterparts. Since $U$ is a deterministic function of $(\phi(h), \psi(r), \phi(t))$ and $\phi$ contains no coverage information beyond frequency, we have $U(h,r,t) \approx U(h',r',t')$ for frequency-matched pairs, giving AUROC $\approx 1/2$. Full proof in Appendix~\ref{app:embedding_proof}. $\square$

\textbf{Why standard KG embeddings satisfy Definition~\ref{def:relation_agnostic}.}
KG embeddings optimize $\min \sum_{(h,r,t)} \mathcal{L}(\text{score}(\phi(h), \psi(r), \phi(t)))$. The loss depends on the \emph{set} of observed triples, not on which specific relations each entity has seen. Since entities with identical frequency profiles receive identical gradient pressure (on average), their embeddings encode frequency patterns but not the combinatorial structure of relation coverage. Consequently:
\textbf{(i)} Variance-based ($\sigma^2_e$): Entity variance reflects training frequency, not relation coverage.
\textbf{(ii)} Deep Ensembles: Each model's $\phi_k(e)$ shares the same blind spot---none encode coverage.
\textbf{(iii)} MC Dropout: Dropout adds noise to $\phi(e)$ but doesn't add coverage information.
\textbf{(iv)} Energy: $E = -\log\sum\exp(f(\phi(h), \psi(r), \phi(t)))$ inherits the embedding's blindness.

\textbf{GNN boundary condition.}
Theorem~\ref{thm:embedding_impossibility} requires \eqref{eq:ra_condition} to hold. GNN architectures can violate this: $\phi(e) = \text{AGG}(\{\phi(n) : n \in \mathcal{N}(e)\})$ aggregates neighbor information, and $\mathcal{N}(e)$ is determined by training graph structure. The impossibility applies when $|\mathcal{R}| \gg \bar{|\mathcal{N}(e)|}$ (relation space exceeds typical neighborhood size). Concretely, on WN18RR with $|\mathcal{R}|{=}11$ and $\bar{|\mathcal{N}(e)|} \approx 2.2$ (ratio $\approx 5$), neighbors CAN proxy for coverage (GNNSafe: 0.79 AUROC). On FB15k-237 with $|\mathcal{R}|{=}237$ and $\bar{|\mathcal{N}(e)|} \approx 19$ (ratio $\approx 12$), neighbors CANNOT (GNNSafe: 0.43 AUROC). Based on these two data points, the transition occurs somewhere in the ratio range 5--12; additional benchmarks would narrow this interval.

\paragraph{Empirical Regularity 1: Complementarity of Uncertainty Signals.}
\label{reg:complementarity}
We observe empirically that semantic and structural uncertainty exhibit predictable, complementary failure modes on \emph{novel-context} and \emph{emerging-entity} OOD (not role-shift OOD, which requires relation-conditioned methods; see \S\ref{sec:relcondvar_exp}):
\textbf{(i)} Semantic uncertainty achieves AUROC $\approx 0.50$ on novel contexts (consistent with Theorem~\ref{thm:impossibility}).
\textbf{(ii)} Structural uncertainty achieves AUROC $< 1$ on emerging entities (some emerging entities have coverage for the query relation).
\textbf{(iii)} The gain from combining signals is dominated by emerging entities:
\begin{equation}
\text{AUROC}(U_{\text{comb}}) - \text{AUROC}(U_{\text{str}}) \approx \pi_e \cdot \left(A_{\text{emerge}}^{\text{comb}}-A_{\text{emerge}}^{\text{str}}\right)
\label{eq:synergy}
\end{equation}
where $\pi_e$ is the emerging-entity fraction. This regularity holds on all 7 datasets without exception (Table~\ref{tab:temporal_ood}). \emph{Interpretation:} When coverage is sufficient ($U_{\text{str}} \approx 1$ on most OOD), semantic adds nothing---as observed on ICEWS14/18. On static benchmarks with partial coverage overlap ($\rho > 0$), semantic breaks ties within coverage strata, providing +8--11pp gains. Eq.~\ref{eq:synergy} matches observation within 0.002 (Appendix~\ref{app:mixture}). \emph{Note:} This is an empirically validated pattern, not a rigorous theorem. Appendix~\ref{app:proof} presents a formalization under idealized assumptions (A1--A6), but A4 is violated on all benchmarks ($\Delta \geq 1.0$). We present this regularity to explain \emph{why} the combination works; it should not be interpreted as providing worst-case guarantees.

\emph{Intuition.} (i) follows from Theorem~\ref{thm:impossibility}. (ii) Emerging entities may have coverage for the query relation, making some indistinguishable from ID via coverage alone. (iii) Since structural AUROC on novel contexts $= 1$ by construction, the novel-context terms cancel and the gain is dominated by emerging entities. See Appendix~\ref{app:proof} for formalization.

\begin{theorem}[Information-Theoretic Characterization]
\label{thm:info_decomp}
Let $I(U; Y)$ denote the mutual information between uncertainty $U$ and OOD indicator $Y$. Let $\rho = P(U_{\text{str}} = 0 | Y_e = 1)$ be the coverage overlap (fraction of emerging entities with coverage for the query relation). Then:
\begin{enumerate}
\item[(i)] $I(U_{\text{str}}; Y_n) = H(Y_n)$: structural uncertainty captures all novel-context information.
\item[(ii)] $I(U_{\text{sem}}; Y_n | U_{\text{str}}) = 0$: semantic adds no information for novel contexts.
\item[(iii)] $I(U_{\text{sem}}; Y_e | U_{\text{str}}) > 0$ iff $\rho > 0$: semantic helps on emerging entities only when coverage overlap exists.
\end{enumerate}
\end{theorem}

\emph{Implication.} This theorem formalizes the decomposition's predictive power: structural is necessary for novel contexts; semantic is necessary for emerging entities only when $\rho > 0$. On static benchmarks ($\rho = 0.34$--$0.66$; Table~\ref{tab:assumptions}), semantic provides +8--11pp gains; on ICEWS14/18 ($\rho \approx 0$), semantic adds nothing---both predicted exactly. Note: (i) and (ii) are definitional consequences of the coverage structure; (iii) is the non-trivial condition determining when semantic uncertainty provides additional detection power. Proof in Appendix~\ref{app:info_proof}.

\subsection{CAGP}
\label{sec:cagp}

We propose CAGP, a minimal realization of the decomposition framework:
\begin{equation}
    U_{\text{CAGP}}(h,r,t) = \alpha \cdot \tilde{U}_{\text{sem}}(h,r,t) + (1-\alpha) \cdot U_{\text{str}}(h,r,t)
    \label{eq:cagp}
\end{equation}
where $\tilde{U}_{\text{sem}}$ is normalized semantic uncertainty and $\alpha{=}0.5$ is a default mixing weight. We use 0.5 because it makes no prior assumption about signal importance; Appendix~\ref{app:alpha_sensitivity} shows AUROC varies $<$0.02 across $\alpha \in [0.05, 0.95]$, making the choice non-critical. Learning $\alpha$ from the link prediction objective is not possible (coverage is discrete and receives no gradient from $\mathcal{L}_{\text{BCE}}$; see Appendix~\ref{app:alpha_sensitivity}). The normalization $\tilde{U}_{\text{sem}} = U_{\text{sem}} \cdot (\bar{U}_{\text{str}} + \epsilon) / (\bar{U}_{\text{sem}} + \epsilon)$ makes semantic uncertainty a rank-preserving tiebreaker \emph{within} coverage strata, yielding +8--11pp emerging-entity gains over $U_{\text{str}}$ alone.

\paragraph{Training.}
Entity embeddings use \emph{reparameterization sampling}~\citep{kingma2014autoencoding}: $\mathbf{e} = \boldsymbol{\mu}_e + \boldsymbol{\sigma}_e \odot \boldsymbol{\epsilon}$, $\boldsymbol{\epsilon} \sim \mathcal{N}(0, I)$, ensuring BCE loss provides variance gradients and preventing KL-driven collapse. The loss combines three terms:
\begin{equation}
\mathcal{L} = \mathcal{L}_{\text{BCE}} + \beta \cdot \mathcal{L}_{\text{KL}} + w_{\text{unc}} \cdot \mathcal{L}_{\text{margin}}
\label{eq:loss}
\end{equation}
with $\beta{=}0.001$, $w_{\text{unc}}{=}0.1$, margin $m{=}0.3$. Ablation confirms $\mathcal{L}_{\text{margin}}$ contributes $<$0.1pp (\S\ref{sec:experiments}).

\paragraph{Coverage matrix.}
We precompute $\mathbf{C} \in \{0,1\}^{|\mathcal{E}| \times |\mathcal{R}|}$ from training data:
$c(e,r) = 1$ iff $\exists (h,r',t) \in \mathcal{T}_{\text{train}}: (e{=}h \lor e{=}t) \land r'{=}r$.
Precomputation is a single pass over training triples ($O(|\mathcal{T}|)$); inference requires two hash lookups per triple ($O(1)$, $<$2\% overhead; Appendix~\ref{app:implementation}).

\paragraph{Design rationale.}
CAGP is intentionally simple---a fixed scalar combination. The contribution is identifying the \emph{necessity} of structural uncertainty (Theorem~\ref{thm:impossibility}) and its \emph{complementarity} with semantic uncertainty (Empirical Regularity~1), not architectural novelty. CAGP serves as a proof-of-concept demonstrating that the decomposition works in practice; the method itself is deliberately minimal because the diagnostic framework---not the architecture---is the contribution.

\paragraph{Alternative realization: RelCondVar.}
\label{sec:relcondvar}
The decomposition framework admits richer realizations. We also evaluate \emph{relation-conditioned variance} (RelCondVar): we train per-relation variance parameters $\sigma^2(e, r)$ alongside entity embeddings, where $\sigma^2(e,r)$ is optimized only on training triples containing entity $e$ with relation $r$. When entity $e$ has never appeared with $r$ (zero coverage), $\sigma^2(e,r)$ remains at its initialization (high uncertainty), naturally recovering the coverage signal in a continuous form. On dense-relation KGs (FB15k-237, 237 relations), RelCondVar outperforms CAGP (+1pp overall, +4pp emerging); on sparse-relation KGs (WN18RR with 11 relations), insufficient per-relation training data prevents learning meaningful conditioned variances, and RelCondVar reduces to $U_{\text{str}}$. This pattern validates the decomposition framework while revealing a scalability axis: more relations $\Rightarrow$ more signal for learned structural uncertainty. \emph{When to use which:} RelCondVar is preferable on dense-relation KGs ($\geq$100 relations) where per-relation variance can be learned reliably; CAGP is preferable on sparse-relation KGs and as a universal default requiring zero hyperparameter tuning. The shared contribution is the decomposition framework itself---both realizations confirm that structural uncertainty is necessary.
