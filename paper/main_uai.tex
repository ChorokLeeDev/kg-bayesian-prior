\section{Theorem Proof}
\label{app:proof}

We state the full assumptions and prove Theorem~\ref{thm:complementarity}.

\paragraph{Assumptions.}
\begin{enumerate}
    \item[\textbf{(A1)}] \textbf{Variance-frequency monotonicity}: $\freq(e_1) > \freq(e_2) \Rightarrow \sigma^2_{e_1} < \sigma^2_{e_2}$.
    \item[\textbf{(A2)}] \textbf{ID coverage}: For all $(h,r,t) \in \mathcal{D}_{\text{ID}}$: $c(h,r) = c(t,r) = 1$.
    \item[\textbf{(A3)}] \textbf{Frequency overlap}: For any $(h,r,t) \in \mathcal{D}_{\text{novel}}$, there exists $(h',r',t') \in \mathcal{D}_{\text{ID}}$ with matching frequencies.
    \item[\textbf{(A4)}] \textbf{Bounded semantic gap}: $\Delta = \max_{x \in \mathcal{D}_{\text{novel}}, x' \in \mathcal{D}_{\text{ID}}} [\tilde{U}_{\text{sem}}(x') - \tilde{U}_{\text{sem}}(x)] < 1$.
    \item[\textbf{(A5)}] \textbf{Non-degenerate coverage}: $0 < \rho = P(U_{\text{str}}(x) = 0 | x \in \mathcal{D}_{\text{emerge}}) < 1$.
    \item[\textbf{(A6)}] \textbf{Semantic separation}: $\tilde{U}_{\text{sem}}(x) > \tilde{U}_{\text{sem}}(x')$ for $x \in \mathcal{D}_{\text{emerge}}, x' \in \mathcal{D}_{\text{ID}}$.
\end{enumerate}

\paragraph{Proof of Part (i).}
By (A3), novel-context entities have frequency-matched ID counterparts. By (A1), matching frequencies imply matching variances. Thus $U_{\text{sem}}$ has identical distributions on novel contexts and ID, giving AUROC $= 0.5$.

\paragraph{Proof of Part (ii).}
By (A2), ID has $U_{\text{str}} = 0$. By (A5), fraction $\rho$ of emerging entities also have $U_{\text{str}} = 0$. AUROC $= P(U_{\text{str}}^{\text{OOD}} > U_{\text{str}}^{\text{ID}}) = 1 - \rho < 1$.

\paragraph{Proof of Part (iii).}
Novel contexts have $c(h,r) = 0$ or $c(t,r) = 0$ by definition, so $U_{\text{str}} \geq 1$. ID has $U_{\text{str}} = 0$ by (A2). Perfect separation gives AUROC $= 1$.

\paragraph{Proof of Part (iv).}
For novel contexts: structural term dominates when $\alpha^* < 1/(1+\Delta)$, preserving perfect separation. For emerging entities: when $U_{\text{str}} > 0$, both terms favor OOD; when $U_{\text{str}} = 0$, semantic term separates by (A6). The combination achieves AUROC $= 1$ on both OOD types, strictly exceeding either signal alone.
