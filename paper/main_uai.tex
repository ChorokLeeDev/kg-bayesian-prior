\section{Implementation Details}
\label{app:implementation}

\paragraph{Hyperparameters.}
Embedding dimension $d=100$, batch size 2048, learning rate $10^{-3}$, KL weight $\beta=0.01$.
All experiments use 50 epochs and are averaged over 3 seeds (42, 123, 456).

\paragraph{Statistical Testing.}
Significance tests ($^\dagger$p$<$0.01) use paired bootstrap resampling with 10,000 iterations.
For each iteration, we sample test triples with replacement and compute AUROC for both methods.
The p-value is the fraction of iterations where the baseline outperforms our method.

\paragraph{Coverage Computation.}
Coverage matrix $\mat{C} \in \{0,1\}^{|\mathcal{E}| \times |\mathcal{R}|}$ is precomputed from training triples.
Storage is sparse: typically $<$5\% of entries are non-zero.
For FB15k-237 (14,541 entities, 237 relations), this requires $<$50MB.

\section{Model Extensions}
\label{app:extensions}

\paragraph{Query-Specific Mixing.}
The attention mechanism uses a 2-layer MLP with hidden dimension 64 and ReLU activation:
\begin{equation}
    \alpha(h,r,t) = \sigma(\text{MLP}([\mathbf{h}; \mathbf{r}; \mathbf{t}; U_{\text{sem}}; U_{\text{str}}]))
\end{equation}
Inputs include entity/relation embeddings and current uncertainty values.

\paragraph{Relation-Conditioned Variance.}
A separate 2-layer MLP predicts variance from entity-relation pairs:
\begin{equation}
    \sigma^2(e,r) = \text{softplus}(\text{MLP}([\mathbf{e}; \mathbf{r}])) + 10^{-6}
\end{equation}
Trained jointly with the KGE objective.

\paragraph{GNN-Based Uncertainty.}
Two message-passing layers with relation-specific attention:
\begin{equation}
    \sigma^{(l+1)}_e = \sigma^{(l)}_e + \sum_{(e',r) \in \mathcal{N}(e)} \alpha_{r} \cdot \sigma^{(l)}_{e'}
\end{equation}
where $\alpha_r$ are learned per-relation attention weights.
