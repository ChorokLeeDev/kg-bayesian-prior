\section{Derivation of Coverage AUROC}
\label{app:proof}

\paragraph{Setup.}
Let $a = P(\text{both covered} | \text{ID})$, $b = P(\text{exactly one covered} | \text{ID})$, $c = 1 - a - b$.
Let $s_r$ be relation sparsity and $q = 1 - s_r$ (probability random entity is covered).

\paragraph{Uncertainty Distributions.}

ID triples:
\begin{align}
    P(U_{\text{ID}} = 0) &= a \\
    P(U_{\text{ID}} = 1) &= b \\
    P(U_{\text{ID}} = 2) &= c
\end{align}

OOD triples (head from ID, random tail):
\begin{align}
    P(U_{\text{OOD}} = 0) &= q \cdot p_h \\
    P(U_{\text{OOD}} = 1) &= q(1-p_h) + (1-q)p_h = q + p_h - 2qp_h \\
    P(U_{\text{OOD}} \geq 1) &\approx s_r \text{ (simplified)}
\end{align}

\paragraph{AUROC Derivation.}

\begin{align}
    \text{AUROC} &= P(U_{\text{ID}} < U_{\text{OOD}}) + \frac{1}{2}P(U_{\text{ID}} = U_{\text{OOD}}) \\
    &= a \cdot s_r + \frac{1}{2}[a(1-s_r) + b \cdot s_r] \\
    &= as_r + \frac{1}{2}a - \frac{1}{2}as_r + \frac{1}{2}bs_r \\
    &= \frac{1}{2}a + \frac{1}{2}as_r + \frac{1}{2}bs_r \\
    &= \frac{1}{2}(a(1+s_r) + s_r \cdot b)
\end{align}

\section{Implementation Details}
\label{app:implementation}

\paragraph{Hyperparameters.}
Embedding dimension $d=100$, batch size 2048, learning rate $10^{-3}$, KL weight $0.01$.
All experiments use 50 epochs and are averaged over 3 seeds (42, 123, 456).

\paragraph{Statistical Testing.}
Significance tests ($^\dagger$p$<$0.01 in tables) use paired bootstrap resampling with 10,000 iterations.
For each iteration, we sample test triples with replacement and compute AUROC for both methods.
The p-value is the fraction of iterations where the baseline outperforms our method.
We report significance only when p$<$0.01 across all three random seeds.

\paragraph{Coverage Computation.}
Coverage matrix $\mat{C} \in \{0,1\}^{|\mathcal{E}| \times |\mathcal{R}|}$ is precomputed from training triples:
\begin{equation}
    C_{e,r} = \mathbb{1}[\exists (h,r,t) \in \mathcal{T} : h=e \lor t=e]
\end{equation}
Storage is sparse: typically $<$5\% of entries are non-zero.

\paragraph{CAGP Implementation.}
\begin{lstlisting}[language=Python]
def cagp_uncertainty(h, r, t, gp_var, coverage, alpha):
    # GP uncertainty (semantic)
    u_gp = (gp_var[h] + gp_var[t]) / 2
    # Coverage uncertainty (structural)
    u_cov = 2.0 - coverage[h, r] - coverage[t, r]
    # Normalize GP to coverage scale
    u_gp_norm = u_gp * u_cov.mean() / u_gp.mean()
    return alpha * u_gp_norm + (1 - alpha) * u_cov
\end{lstlisting}

\paragraph{WN18RR Structural Properties.}
WN18RR presents a challenging setting for coverage-based methods due to its sparse relation structure:
\begin{itemize}
    \item \textbf{Few relations}: Only 11 relation types means each relation covers many entities (avg 7,894 triples/relation vs 1,148 for FB15k-237)
    \item \textbf{High entity-relation overlap}: 78\% of entities appear with 3+ different relations, versus 45\% for FB15k-237
    \item \textbf{Hierarchical structure}: WordNet's taxonomy creates dense connectivity---most entities have paths to common hypernyms
\end{itemize}

These properties explain why coverage alone achieves only 0.657 AUROC on WN18RR (versus 0.821 on FB15k-237): the binary signal is less discriminative when most entities are ``covered'' for most relations.
The GP signal becomes more valuable in this setting, as reflected in the learned $\alpha = 0.52$ (balanced weighting).

\section{Calibration Details}
\label{app:calibration}

\paragraph{ECE Computation.}
Expected Calibration Error is computed by binning predictions into 10 equal-width bins based on predicted uncertainty, then measuring the gap between predicted and actual OOD rates:
\begin{equation}
    \text{ECE} = \sum_{b=1}^{10} \frac{|B_b|}{N} \left| \text{acc}(B_b) - \text{conf}(B_b) \right|
\end{equation}
where $B_b$ is the set of samples in bin $b$, $\text{acc}(B_b)$ is the fraction of correct predictions (ID samples), and $\text{conf}(B_b)$ is the mean predicted confidence (1 - uncertainty).
