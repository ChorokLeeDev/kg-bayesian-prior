\subsection{Preliminaries}

\textbf{Frequency}: $\freq(e)$ counts triples containing entity $e$.
\textbf{Coverage}: $c(e,r) \in \{0,1\}$ indicates whether $e$ appears with $r$ in training.

\begin{definition}[OOD Partition]
\label{def:ood}
Fix threshold $\tau > 0$. OOD queries partition into:
\textbf{Emerging entities} $\mathcal{D}_{\text{emerge}}$: $\min(\freq(h), \freq(t)) \leq \tau$.
\textbf{Novel contexts} $\mathcal{D}_{\text{novel}}$: $\min(\freq(h), \freq(t)) > \tau$ and ($c(h,r)=0$ or $c(t,r)=0$).
We set $\tau$ to the 25th percentile of entity frequencies.
\end{definition}

Novel-context detection by $U_{\text{str}}$ is perfect by construction (the same $c(e,r)$ defines both the OOD label and the detector). The contributions are: (1) proving all embedding-based methods are blind to novel contexts (Theorem~\ref{thm:impossibility}), (2) demonstrating this empirically---top-confident predictions contain zero-evidence queries at 2--2.5$\times$ the base rate (Table~\ref{tab:confident_wrong}), and (3) quantifying that 11--25\% of test queries fall into this blind spot. See Remark~\ref{rem:circularity}.

\subsection{Uncertainty Signals}

\textbf{Semantic uncertainty} (entity embedding variance): $U_{\text{sem}}(h,r,t) = \frac{1}{2}(\bar{\sigma}^2_h + \bar{\sigma}^2_t)$ where $\bar{\sigma}^2_e$ is the mean diagonal variance. This is \emph{relation-agnostic}: it depends only on entity variances, not the query relation.

\textbf{Structural uncertainty} (coverage): $U_{\text{str}}(h,r,t) = 2 - c(h,r) - c(t,r) \in \{0, 1, 2\}$.

\subsection{Theoretical Results}

\begin{theorem}[Limitation of Relation-Agnostic Detection]
\label{thm:impossibility}
Let $U(h,r,t) = f(\sigma^2_h, \sigma^2_t)$ where $\sigma^2_e$ depends only on entity $e$, and $f$ is any combining function. Under assumptions A1--A3 (Appendix~\ref{app:proof}):
\begin{equation}
    \text{AUROC}(U, \mathcal{D}_{\text{novel}}) \leq \frac{1}{2} + O(\epsilon)
\end{equation}
where $\epsilon$ measures frequency-matching tightness (tolerance $\epsilon{=}10$ achieves $\geq$98\% matching on all datasets). \emph{Assumptions validated}: Spearman $\rho_s(\text{freq}, \sigma^2) \leq {-}0.68$ on all datasets (A1); $\geq$98\% of novel-context triples have frequency-matched ID counterparts within $\epsilon{=}10$ (A3).
\end{theorem}

\textbf{Proof sketch.} By A1, $\sigma^2_e$ is determined by $\text{freq}(e)$. By A3, novel-context triples have frequency-matched ID counterparts, so $U(h,r,t) \approx U(h',r',t')$, giving AUROC $\approx 1/2$. Full proof in Appendix~\ref{app:proof}.

\begin{definition}[Relation-Agnostic Embedding (Sufficient Condition)]
\label{def:relation_agnostic}
An embedding $\phi: \mathcal{E} \to \mathbb{R}^d$ is \emph{relation-agnostic} if $I(\phi(e); c(e,r) \mid \freq(e)) = 0$ for all $r$. This is a \emph{sufficient} condition for the limitation---embeddings violating it may still fail if standard uncertainty methods cannot extract the coverage signal.
\end{definition}

\begin{theorem}[Embedding-Based Limitation]
\label{thm:embedding_impossibility}
Let $\phi$ satisfy Definition~\ref{def:relation_agnostic}. For any $U(h,r,t) = F(\phi(h), \psi(r), \phi(t))$: $\text{AUROC}(U, \mathcal{D}_{\text{novel}}) \leq 1/2 + O(\epsilon)$.
\end{theorem}

\textbf{Key insight.} Real embeddings violate Definition~\ref{def:relation_agnostic}---trained DistMult achieves AUC$=$0.96 for predicting coverage within frequency strata, meaning the CMI condition fails. Yet novel-context AUROC remains 0.34--0.50 (Table~\ref{tab:complementarity}). Why? Coverage information exists in embeddings but is \emph{architecturally inaccessible}: standard uncertainty methods (variance, energy, ensembles) cannot extract it without explicit supervision. This explains why the limitation holds in practice even when the sufficient condition is violated. Coverage-aware training fails to generalize: AUROC on held-out unseen pairs \emph{decreases} from 0.63 to 0.58 (Appendix~\ref{app:coverage_training}). This confirms explicit coverage tracking is necessary.

\paragraph{Empirical Regularity 1: Complementarity.}
\label{reg:complementarity}
Semantic and structural uncertainty exhibit complementary failure modes:
\textbf{(i)} Semantic achieves AUROC $\approx 0.50$ on novel contexts (Theorem~\ref{thm:impossibility}).
\textbf{(ii)} Structural achieves AUROC $< 1$ on emerging entities (some have coverage).
\textbf{(iii)} The combination gain is dominated by emerging entities:
\begin{equation}
\text{AUROC}(U_{\text{comb}}) - \text{AUROC}(U_{\text{str}}) \approx \pi_e \cdot \left(A_{\text{emerge}}^{\text{comb}}-A_{\text{emerge}}^{\text{str}}\right)
\label{eq:synergy}
\end{equation}
This holds on all datasets tested (Table~\ref{tab:temporal_ood}, Appendix~\ref{app:mixture}); Eq.~\ref{eq:synergy} matches observation within 0.002.

\begin{theorem}[Information-Theoretic Characterization]
\label{thm:info_decomp}
Let $\rho = P(U_{\text{str}} = 0 | Y_e = 1)$ be coverage overlap. Then:
(i) $I(U_{\text{str}}; Y_n) = H(Y_n)$: structural captures all novel-context information.
(ii) $I(U_{\text{sem}}; Y_n | U_{\text{str}}) = 0$: semantic adds nothing for novel contexts.
(iii) $I(U_{\text{sem}}; Y_e | U_{\text{str}}) > 0$ iff $\rho > 0$: semantic helps on emerging entities only when coverage overlap exists.
\end{theorem}

On static benchmarks ($\rho = 0.34$--$0.66$), semantic provides +8--11pp gains; on ICEWS14/18 ($\rho \approx 0$), semantic adds nothing---both predicted exactly.

\subsection{CAGP}
\label{sec:cagp}

We propose CAGP:
\begin{equation}
    U_{\text{CAGP}}(h,r,t) = \alpha \cdot \tilde{U}_{\text{sem}}(h,r,t) + (1-\alpha) \cdot U_{\text{str}}(h,r,t)
    \label{eq:cagp}
\end{equation}
where $\tilde{U}_{\text{sem}}$ is normalized semantic uncertainty and $\alpha{=}0.5$. AUROC varies $<$0.02 across $\alpha \in [0.05, 0.95]$ (Appendix~\ref{app:alpha_sensitivity}).

\paragraph{Training.}
Entity embeddings use reparameterization sampling: $\mathbf{e} = \boldsymbol{\mu}_e + \boldsymbol{\sigma}_e \odot \boldsymbol{\epsilon}$. Loss: $\mathcal{L} = \mathcal{L}_{\text{BCE}} + 0.001 \cdot \mathcal{L}_{\text{KL}} + 0.1 \cdot \mathcal{L}_{\text{margin}}$.

\paragraph{Coverage matrix.}
We precompute $\mathbf{C} \in \{0,1\}^{|\mathcal{E}| \times |\mathcal{R}|}$ from training data in $O(|\mathcal{T}|)$. Inference: batched tensor lookup adds negligible overhead. For billion-scale KGs, Bloom filters at 1\% FPR reduce storage 5$\times$ with $<$0.2pp recall degradation (Appendix~\ref{app:implementation}).

\paragraph{Design rationale.}
CAGP is intentionally simple---a hash table lookup combined with learned variance. The contribution is \emph{not} architectural novelty but rather: (1) identifying that \emph{no} existing uncertainty method tracks coverage, despite its necessity (Theorem~\ref{thm:impossibility}), (2) quantifying that this affects 11--25\% of queries, and (3) showing the combination with semantic uncertainty is complementary (Empirical Regularity~1). The fix is obvious in hindsight; the discovery is that it was universally missing.
